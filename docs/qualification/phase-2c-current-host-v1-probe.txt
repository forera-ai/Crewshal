from pathlib import Path

assert hashlib.sha256(hlt.encode()).hexdigest() == "cfce2653fe58d2a53c7ebc5a541b3067737c2e788ac662183a014807b30b4a2f"
def sample():
    mem = dict(line.split(":", 1) for line in Path("/proc/meminfo").read_text().splitlines())
    disk = shutil.disk_usage("/srv")
    return {"available_memory_bytes": int(mem["MemAvailable"].split()[0])*1024,
            "total_memory_bytes": int(mem["MemTotal"].split()[0])*1024,
            "swap_total_bytes": int(mem["SwapTotal"].split()[0])*1024,
            "free_disk_bytes": disk.free}
result = {"scope":"trusted current-host diagnostic only", "execution_allowed":False,
          "os_release":Path("/etc/os-release").read_text(), "kernel":platform.release(),
          "architecture":platform.machine(), "initial_resources":sample()}
result["cgroup"] = {name:Path("/sys/fs/cgroup",name).read_text() for name in ("cgroup.controllers","cgroup.subtree_control")}
result["mounts"] = subprocess.run(["findmnt","--json","-T","/srv","-o","FSTYPE,OPTIONS"],capture_output=True,text=True,timeout=5).stdout
result["tools"] = {name:shutil.which(name) for name in ("python3","systemd-run","timeout","xfs_quota","mkfs.xfs","mkfs.ext4","mount","losetup","docker","sbx")}
result["hlt_runs"] = []
for i in range(2):
    proc = subprocess.run(["python3","-c",hlt],capture_output=True,text=True,timeout=10,env={"PATH":"/usr/bin:/bin"})
    assert len(proc.stdout.encode())<=65536 and len(proc.stderr.encode())<=65536
    result["hlt_runs"].append({"exit_code":proc.returncode,"stdout":proc.stdout,"stderr":proc.stderr})
result["final_resources"] = sample()
handles=[]
unreadable=[]
for proc in Path("/proc").iterdir():
    if not proc.name.isdigit(): continue
    try:
        for fd in (proc/"fd").iterdir():
            try: target=os.readlink(fd)
            except FileNotFoundError: continue
            if "kvm-vm" in target or "kvm-vcpu" in target: handles.append({"pid":int(proc.name),"target":target})
    except FileNotFoundError: pass
    except PermissionError: unreadable.append(int(proc.name))
result["final_kvm_handles"]=handles
result["unreadable_proc_entries"]=unreadable
print(json.dumps(result,sort_keys=True))
